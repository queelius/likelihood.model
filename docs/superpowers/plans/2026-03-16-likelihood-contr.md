> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create the `likelihood.contr` R package: a contribution builder for constructing likelihood models from heterogeneous observation types.

**Architecture:** Three S3 constructors (`contr_name`, `contr_fn`, `likelihood_contr`) compose "contribution" objects into a `likelihood_model`. No R6. Dispatch by column name (string) or custom function. Per-contribution numerical fallback for derivatives.

**Tech Stack:** R, S3 classes, numDeriv, testthat 3

**Spec:** `docs/superpowers/specs/2026-03-16-v1.0-and-likelihood.contr-design.md` (Package 2 section)

#' Create a contribution from custom functions
#'
#' Wraps user-provided log-likelihood (and optional derivative) functions
#' into a contribution object for use with [likelihood_contr()].
#'
#' @param loglik A function(df, par, ...) computing the log-likelihood
#'   contribution for a subset of data.
#' @param score Optional function(df, par, ...) computing the score.
#'   Falls back to numerical differentiation via numDeriv::grad.
#' @param hess Optional function(df, par, ...) computing the Hessian.
#'   Falls back to numerical differentiation via numDeriv::hessian.
#' @return An S3 object of class "contr"
#' @export
contr_fn <- function(loglik, score = NULL, hess = NULL) {
  if (missing(loglik) || !is.function(loglik)) {
    stop("loglik must be a function(df, par, ...)")
  }
  if (!is.null(score) && !is.function(score)) {
    stop("score must be a function(df, par, ...) or NULL")
  }
  if (!is.null(hess) && !is.function(hess)) {
    stop("hess must be a function(df, par, ...) or NULL")
  }
  structure(
    list(loglik = loglik, score = score, hess = hess),
    class = "contr"
  )
}
```

#' Create a contribution from a named R distribution
#'
#' Generates a log-likelihood contribution for a specific observation type
#' using standard R distribution functions (d<name>, p<name>).
#'
#' @param dist_name Distribution name (e.g., "weibull", "norm", "exp")
#' @param type One of "exact", "right", "left", "interval"
#' @param ob_col Column name for observations (default "x")
#' @param ob_col_upper Column name for upper bound (interval censoring only)
#' @return An S3 object of class "contr"
#' @export
contr_name <- function(dist_name, type, ob_col = "x", ob_col_upper = NULL) {
  stopifnot(is.character(dist_name), length(dist_name) == 1)
  type <- match.arg(type, c("exact", "right", "left", "interval"))

  if (type == "interval" && is.null(ob_col_upper)) {
    stop("ob_col_upper is required for interval censoring")
  }

  pdf_name <- paste0("d", dist_name)
  cdf_name <- paste0("p", dist_name)
  if (!exists(pdf_name, mode = "function")) {
    stop(sprintf("Distribution '%s' not found: no function '%s'", dist_name, pdf_name))
  }
  if (!exists(cdf_name, mode = "function")) {
    stop(sprintf("Distribution '%s' not found: no function '%s'", dist_name, cdf_name))
  }

  pdf <- get(pdf_name, mode = "function")
  cdf <- get(cdf_name, mode = "function")

  ll <- switch(type,
    exact = function(df, par, ...) {
      xs <- df[[ob_col]]
      if (is.null(xs)) stop(sprintf("Column '%s' not found in data frame", ob_col))
      par_args <- .prepare_args_list(par, pdf)
      sum(do.call(pdf, c(list(xs), par_args, log = TRUE)))
    },
    right = function(df, par, ...) {
      xs <- df[[ob_col]]
      if (is.null(xs)) stop(sprintf("Column '%s' not found in data frame", ob_col))
      par_args <- .prepare_args_list(par, cdf)
      sum(do.call(cdf, c(list(xs), par_args, log.p = TRUE, lower.tail = FALSE)))
    },
    left = function(df, par, ...) {
      xs <- df[[ob_col]]
      if (is.null(xs)) stop(sprintf("Column '%s' not found in data frame", ob_col))
      par_args <- .prepare_args_list(par, cdf)
      sum(do.call(cdf, c(list(xs), par_args, log.p = TRUE)))
    },
    interval = function(df, par, ...) {
      xs_lower <- df[[ob_col]]
      xs_upper <- df[[ob_col_upper]]
      if (is.null(xs_lower)) stop(sprintf("Column '%s' not found in data frame", ob_col))
      if (is.null(xs_upper)) stop(sprintf("Column '%s' not found in data frame", ob_col_upper))
      par_args <- .prepare_args_list(par, cdf)
      sum(log(
        do.call(cdf, c(list(xs_upper), par_args)) -
        do.call(cdf, c(list(xs_lower), par_args))
      ))
    }
  )

  contr_fn(loglik = ll)
}


#' Map parameter vector to distribution function arguments
#' @param par Parameter vector (named or unnamed)
#' @param fn Distribution function to match against
#' @return Named list of parameters
#' @keywords internal
.prepare_args_list <- function(par, fn) {
  fn_args <- names(formals(fn))
  fn_args <- fn_args[!fn_args %in% c("x", "q", "p", "n", "log", "log.p",
                                      "lower.tail", "...")]
  if (length(par) > length(fn_args)) {
    stop(sprintf(
      "Too many parameters (%d) for distribution (expects at most %d: %s)",
      length(par), length(fn_args), paste(fn_args, collapse = ", ")
    ))
  }
  if (is.null(names(par))) {
    names(par) <- fn_args[seq_along(par)]
  }
  as.list(par)
}
```

#' Create a new cache environment for likelihood_contr closures
#' @return An environment with df_ref, dfs, and fns slots
#' @keywords internal
.new_cache <- function() {
  env <- new.env(parent = emptyenv())
  env$df_ref <- NULL
  env$dfs <- NULL
  env
}

#' Populate or reuse the cache for the given data frame
#'
#' Splits the data frame by observation type if not already cached.
#'
#' @param cache Cache environment from .new_cache()
#' @param df Data frame
#' @param model A likelihood_contr model
#' @return The cache environment with $dfs populated
#' @keywords internal
.ensure_cache <- function(cache, df, model) {
  if (!identical(df, cache$df_ref)) {
    # Determine types
    if (!is.null(model$obs_type_col)) {
      types <- df[[model$obs_type_col]]
      if (is.null(types)) {
        stop(sprintf("Column '%s' not found in data frame", model$obs_type_col))
      }
    } else {
      types <- model$obs_type_fn(df)
    }

    if (any(is.na(types))) {
      stop("obs_type produced NA values; all observations must have a type")
    }

    unmatched <- setdiff(unique(types), names(model$contributions))
    if (length(unmatched) > 0) {
      stop(sprintf(
        "No contribution registered for type(s): %s",
        paste(sQuote(unmatched), collapse = ", ")
      ))
    }

    cache$df_ref <- df
    cache$dfs <- split(df, types)
  }
  cache
}

#' Compose contributions into a likelihood model
#'
#' Combines named contribution objects into a single likelihood model.
#' The resulting model splits data by observation type, evaluates each
#' contribution's log-likelihood on its subset, and sums.
#'
#' @param obs_type Either a string (column name to dispatch on) or a
#'   function(df) returning a character vector of type labels.
#' @param ... Named contr objects. Names must match obs_type values.
#' @param rdata_fn Optional function(theta, n, ...) for random data
#'   generation. Required for fim() to work.
#' @param assumptions Character vector of assumptions (iid always included).
#' @return An S3 object of class c("likelihood_contr", "likelihood_model")
#' @importFrom likelihood.model loglik score hess_loglik assumptions rdata
#'   is_likelihood_model fim
#' @export
likelihood_contr <- function(obs_type, ..., rdata_fn = NULL,
                              assumptions = "iid") {
  contributions <- list(...)

  if (length(contributions) == 0) {
    stop("likelihood_contr requires at least one named contribution")
  }
  if (is.null(names(contributions)) || any(names(contributions) == "")) {
    stop("All contributions must be named")
  }
  for (nm in names(contributions)) {
    if (!inherits(contributions[[nm]], "contr")) {
      stop(sprintf("Contribution '%s' must be a contr object (from contr_fn or contr_name)", nm))
    }
  }

  obs_type_col <- NULL
  obs_type_fn <- NULL
  if (is.character(obs_type) && length(obs_type) == 1) {
    obs_type_col <- obs_type
  } else if (is.function(obs_type)) {
    obs_type_fn <- obs_type
  } else {
    stop("obs_type must be a column name (string) or a function(df)")
  }

  structure(
    list(
      obs_type_col = obs_type_col,
      obs_type_fn = obs_type_fn,
      contributions = contributions,
      rdata_fn = rdata_fn,
      assumptions = unique(c("iid", assumptions))
    ),
    class = c("likelihood_contr", "likelihood_model")
  )
}

#' @export
assumptions.likelihood_contr <- function(model, ...) {
  model$assumptions
}

#' @export
loglik.likelihood_contr <- function(model, ...) {
  cache <- .new_cache()
  contribs <- model$contributions
  p_len <- NULL  # lazily determined

  function(df, par, ...) {
    cache <- .ensure_cache(cache, df, model)
    total <- 0
    for (type in names(cache$dfs)) {
      sub_df <- cache$dfs[[type]]
      if (nrow(sub_df) == 0) next
      total <- total + contribs[[type]]$loglik(sub_df, par, ...)
    }
    total
  }
}

#' @importFrom numDeriv grad
#' @export
score.likelihood_contr <- function(model, ...) {
  cache <- .new_cache()
  contribs <- model$contributions

  function(df, par, ...) {
    cache <- .ensure_cache(cache, df, model)
    p <- length(par)
    total <- rep(0, p)
    for (type in names(cache$dfs)) {
      sub_df <- cache$dfs[[type]]
      if (nrow(sub_df) == 0) next
      ctr <- contribs[[type]]
      if (!is.null(ctr$score)) {
        total <- total + ctr$score(sub_df, par, ...)
      } else {
        ll <- ctr$loglik
        total <- total + numDeriv::grad(
          func = function(p) ll(sub_df, p, ...),
          x = par, method.args = list(r = 6)
        )
      }
    }
    total
  }
}

#' @importFrom numDeriv hessian
#' @export
hess_loglik.likelihood_contr <- function(model, ...) {
  cache <- .new_cache()
  contribs <- model$contributions

  function(df, par, ...) {
    cache <- .ensure_cache(cache, df, model)
    p <- length(par)
    total <- matrix(0, p, p)
    for (type in names(cache$dfs)) {
      sub_df <- cache$dfs[[type]]
      if (nrow(sub_df) == 0) next
      ctr <- contribs[[type]]
      if (!is.null(ctr$hess)) {
        total <- total + ctr$hess(sub_df, par, ...)
      } else {
        ll <- ctr$loglik
        total <- total + numDeriv::hessian(
          func = function(p) ll(sub_df, p, ...),
          x = par, method.args = list(r = 6)
        )
      }
    }
    total
  }
}
